internal sealed class Torneko4Package
{
    public TornekoBinPreservationSnapshot Snapshot { get; }
    public byte[] EditedBase { get; }
    public string OriginalSourcePath { get; }
    public System.Collections.Generic.List<Torneko4MemoryPatch> Patches { get; }
    public System.Collections.Generic.Dictionary<int, ushort> MonsterActions { get; }

    public Torneko4Package(
        TornekoBinPreservationSnapshot snapshot,
        byte[] editedBase,
        string originalSourcePath,
        System.Collections.Generic.List<Torneko4MemoryPatch> patches = null,
        System.Collections.Generic.Dictionary<int, ushort> monsterActions = null)
    {
        Snapshot = snapshot;
        EditedBase = (byte[])editedBase.Clone();
        OriginalSourcePath = originalSourcePath ?? "";

        Patches = patches == null
            ? new System.Collections.Generic.List<Torneko4MemoryPatch>()
            : patches.ConvertAll(x => x.Clone());

        MonsterActions = monsterActions == null
            ? new System.Collections.Generic.Dictionary<int, ushort>()
            : new System.Collections.Generic.Dictionary<int, ushort>(monsterActions);
    }
}

    public static void Save(
        string path,

        byte[] editedBase,
        System.Collections.Generic.IEnumerable<Torneko4MemoryPatch> patches,
        System.Collections.Generic.IDictionary<int, ushort> monsterActions,
        System.Collections.Generic.IDictionary<int, bool> ganirasuBarrier = null)
    {
        if (snapshot == null)
            throw new ArgumentNullException(nameof(snapshot));

                "編集済み基本領域のサイズが不正です。");

        if (string.IsNullOrWhiteSpace(path))
            throw new ArgumentException("保存先が指定されていません。");

            original.Length > MaximumOriginalSize)
        {
            throw new InvalidDataException(
                "元データのサイズが保存可能範囲外です。");

            TornekoBinPreservation.ComputeSha256(original);

            StringComparison.OrdinalIgnoreCase))

                        "追加変更データのサイズが不正です。");

            if (check.Patches.Count != patchList.Count)

                    "保存後の追加変更数検証に失敗しました。");

            if (check.MonsterActions.Count != actionList.Count)

            foreach (var pair in actionList)

                    throw new IOException(
                        "保存後の追加変更データ検証に失敗しました。");
                }
            }
            File.Move(tempPath, fullPath, true);

        finally

            if (File.Exists(tempPath))

    public static Torneko4Package Load(string path)
    {
        if (string.IsNullOrWhiteSpace(path) || !File.Exists(path))
            throw new FileNotFoundException(
                "Torneko4ファイルが見つかりません。",
                path);

        FileInfo info = new FileInfo(path);

        if (info.Length < 100 ||
            info.Length > MaximumPackageSize)
        {
            throw new InvalidDataException(
                "Torneko4ファイルのサイズが不正です。");
        }

        using var stream = new FileStream(
            path,
            FileMode.Open,
            FileAccess.Read,
            FileShare.Read);

        using var reader = new BinaryReader(
            stream,
            System.Text.Encoding.UTF8,
            true);

        byte[] magic = reader.ReadBytes(8);

        if (!System.Linq.Enumerable.SequenceEqual(magic, Magic))
            throw new InvalidDataException(
                "Torneko4識別ヘッダーが一致しません。");

        int version = reader.ReadInt32();

        if (version != 1 && version != 2 && version != 3 && version != FormatVersion)
            throw new InvalidDataException(
                "未対応のTorneko4バージョンです: " + version);

        int originalLength = reader.ReadInt32();
        int editedLength = reader.ReadInt32();

        if (originalLength < BaseSize ||
            originalLength > MaximumOriginalSize ||
            editedLength != BaseSize)
        {
            throw new InvalidDataException(
                "Torneko4内の基本データサイズが不正です。");
        }

        string originalSourcePath = reader.ReadString();
        string expectedOriginalHash = reader.ReadString();
        string expectedEditedHash = reader.ReadString();

        if (expectedOriginalHash.Length != 64 ||
            expectedEditedHash.Length != 64)
        {
            throw new InvalidDataException(
                "Torneko4のハッシュ情報が不正です。");
        }

        byte[] original = reader.ReadBytes(originalLength);
        byte[] editedBase = reader.ReadBytes(editedLength);

        if (original.Length != originalLength ||
            editedBase.Length != editedLength)
        {
            throw new InvalidDataException(
                "Torneko4のデータが途中で切れています。");
        }

        string actualOriginalHash =
            TornekoBinPreservation.ComputeSha256(original);

        string actualEditedHash =
            TornekoBinPreservation.ComputeSha256(editedBase);

        if (!string.Equals(
            expectedOriginalHash,
            actualOriginalHash,
            StringComparison.OrdinalIgnoreCase))
        {
            throw new InvalidDataException(
                "元データのSHA256が一致しません。");
        }

        if (!string.Equals(
            expectedEditedHash,
            actualEditedHash,
            StringComparison.OrdinalIgnoreCase))
        {
            throw new InvalidDataException(
                "編集データのSHA256が一致しません。");
        }

        var patchList =
            new System.Collections.Generic.List<Torneko4MemoryPatch>();

        // バージョン1は従来どおり追加領域なし。
        if (version >= 2)
        {
            int patchCount = reader.ReadInt32();

            if (patchCount < 0 || patchCount > MaximumPatchCount)
                throw new InvalidDataException(
                    "追加変更の件数が不正です。");

            long totalPatchBytes = 0;

            for (int i = 0; i < patchCount; i++)
            {
                string name = reader.ReadString();
                long address = reader.ReadInt64();
                int dataLength = reader.ReadInt32();
                string expectedPatchHash = reader.ReadString();

                if (string.IsNullOrWhiteSpace(name) ||
                    dataLength <= 0 ||
                    dataLength > MaximumPatchSize ||
                    expectedPatchHash.Length != 64)
                {
                    throw new InvalidDataException(
                        "追加変更の情報が不正です。");
                }

                totalPatchBytes += dataLength;

                if (totalPatchBytes > MaximumPackageSize)
                    throw new InvalidDataException(
                        "追加変更データが上限を超えています。");

                byte[] data = reader.ReadBytes(dataLength);

                if (data.Length != dataLength)
                    throw new InvalidDataException(
                        "追加変更データが途中で切れています。");

                string actualPatchHash =
                    TornekoBinPreservation.ComputeSha256(data);

                if (!string.Equals(
                    expectedPatchHash,
                    actualPatchHash,
                    StringComparison.OrdinalIgnoreCase))
                {
                    throw new InvalidDataException(
                        "追加変更データのSHA256が一致しません: " + name);
                }

                patchList.Add(
                    new Torneko4MemoryPatch(name, address, data));
            }
        }

        var monsterActions =
            new System.Collections.Generic.Dictionary<int, ushort>();

        // バージョン3で追加されたモンスター別特技設定。
        if (version >= 3)
        {
            int actionCount = reader.ReadInt32();

            if (actionCount < 0 || actionCount > 199)
                throw new InvalidDataException(
                    "モンスター別特技設定の件数が不正です。");

            for (int i = 0; i < actionCount; i++)
            {
                int monsterId = reader.ReadInt32();
                ushort actionId = reader.ReadUInt16();

                if (monsterId < 0 || monsterId >= 199)
                    throw new InvalidDataException(
                        "モンスター番号が不正です: " + monsterId);

                if (actionId > 0xFF)
                    throw new InvalidDataException(
                        "特技IDが不正です: " + actionId);

                if (monsterActions.ContainsKey(monsterId))
                    throw new InvalidDataException(
                        "モンスター別特技設定が重複しています: " + monsterId);

                monsterActions.Add(monsterId, actionId);
            }
        }

        if (stream.Position != stream.Length)
            throw new InvalidDataException(
                "Torneko4の末尾に不明なデータがあります。");

        string embeddedPath = Path.Combine(
            Path.GetTempPath(),
            "Torneko4Embedded_" +
            Guid.NewGuid().ToString("N") +
            ".bin");

        var snapshot = new TornekoBinPreservationSnapshot(
            original,
            embeddedPath);

        return new Torneko4Package(
            snapshot,
            editedBase,
            originalSourcePath,
            patchList,
            monsterActions);
    }

// ============================================================
// メインメニューとPCSX2接続画面で共有する状態。
// 接続とBIN読み込みを別々に管理する。
// ============================================================

internal static class Pcsx2LiveSession
{
    public static bool IsConnected { get; private set; }
    public static bool VanillaDataLoaded { get; private set; }
    public static int ProcessId { get; private set; }
    public static byte[]? VanillaBin { get; private set; }
    public static Pcsx2MemoryLayout? MemoryLayout { get; private set; }

    public static event System.Action? StateChanged;

    public static void SetConnected(int processId)
    {
        IsConnected = true;
        ProcessId = processId;

        // 新しい接続では新しい取得を待つ。
        VanillaDataLoaded = false;
        VanillaBin = null;
        MemoryLayout = null;

        StateChanged?.Invoke();
    }

    public static void SetVanillaData(byte[] data)
    {
        if (data == null || data.Length != 524288)
            throw new System.ArgumentException(
                "取得BINは524288 bytesである必要があります。");

        VanillaBin = (byte[])data.Clone();
        VanillaDataLoaded = true;

        StateChanged?.Invoke();
    }

    public static void SetMemoryLayout(Pcsx2MemoryLayout layout)
    {
        if (layout == null)
            throw new System.ArgumentNullException(nameof(layout));

        MemoryLayout = layout;
        StateChanged?.Invoke();
    }
    public static void SetDisconnected()
    {
        IsConnected = false;
        ProcessId = 0;
        MemoryLayout = null;

        // 取得済みBINはメイン画面の編集バッファに残す。
        StateChanged?.Invoke();
    }
}
