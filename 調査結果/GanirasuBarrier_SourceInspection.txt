  847:     private static bool TryResolveRecordTable(
  848:         byte[]? bin,
  849:         out int tableOffset,
  850:         out string reason)
  851:     {
  852:         tableOffset = -1;
  853:         reason = "BINファイルを読み込むと基本能力値を編集できます。";
  854: 
  855:         if (bin == null)
  856:             return false;
  857: 
  858:         if (bin.Length < BinFileService.ExpectedSize)
  859:         {
  860:             reason = "BINサイズが524,288 bytes未満のため編集できません。";
  861:             return false;
  862:         }
  863: 
  864:         // 旧HSP label_073:
  865:         // var_142 = peek(var_134, 3)
  866:         // var_143 = var_142 * 0x01000000
  867:         // var_149 = DWORD(var_134 + 16) + var_143
  868:         uint rawPointer = BinFileService.ReadUInt32(bin, 16);
  869:         uint correction = (uint)bin[3] << 24;
  870:         uint pointer = unchecked(rawPointer + correction);
  871: 
  872:         // 旧版のEEアドレス 0x20180000 を物理RAM側へ正規化。
  873:         uint physicalAddress = pointer & EeAddressMask;
  874: 
  875:         if (physicalAddress < EeMemoryBaseAddress)
  876:         {
  877:             reason = "モンスターデータへのポインタが不正です。BINの種類を確認してください。";
  878:             return false;
  879:         }
  880: 
  881:         long offset = (long)physicalAddress - EeMemoryBaseAddress;
  882: 
  883:         // モンスターIDとBINレコードの対応を1レコード分補正
  884: 
  885:         offset += MonsterRecordSize;
  886:         long requiredEnd = offset + (long)MonsterRecordCount * MonsterRecordSize;
  887: 
  888:         if (offset < 0 || requiredEnd > bin.Length)
  889:         {
  890:             reason =
  891:                 "モンスターデータ領域がBIN内にありません。\r\n" +
  892:                 "推定位置：0x" + offset.ToString("X") +
  893:                 " ／ 必要長：" + (MonsterRecordCount * MonsterRecordSize) + " bytes";
  894:             return false;
  895:         }
  896: 
  897:         tableOffset = checked((int)offset);
  898:         reason = "";
  899:         return true;
  900:     }

  935:     private static uint ReadUInt32LE(byte[] data, int offset)
  936:     {
  937:         BinFileService.CheckRange(data, offset, 4);
  938:         return (uint)data[offset]
  939:             | ((uint)data[offset + 1] << 8)
  940:             | ((uint)data[offset + 2] << 16)
  941:             | ((uint)data[offset + 3] << 24);
  942:     }
  943: 
  944:     private static void WriteUInt32LE(byte[] data, int offset, uint value)
  945:     {
  946:         BinFileService.CheckRange(data, offset, 4);
  947:         data[offset] = (byte)value;
  948:         data[offset + 1] = (byte)(value >> 8);
  949:         data[offset + 2] = (byte)(value >> 16);
  950:         data[offset + 3] = (byte)(value >> 24);
  951:     }
